# LeKiwi 抓球动作配置。修改后重启程序即可生效，不需要重新编译。
# 姿态角度均为标定后的度数，不是舵机 0～4095 原始值。

# 电机 ID：1肩部水平，2肩部抬升，3肘部，4腕部俯仰，5腕部旋转，
#          6夹爪，7左轮，8后轮，9右轮。

# 夹球最终姿态：四项位置偏移均为0时，ID1～ID5到达这里后ID6闭合。
# ID4是夹球瞬间的直接目标角度，不再由“固定总俯仰角”反算。程序会自动
# 计算安全PRE_GRAB姿态，再平滑过渡到下面五个最终角度。
# 这里暂时使用当前(0.120m, -0.060m)抓取点对应的角度；实机确认新的夹球
# 姿态后，只替换下面五个值，不要把ID6的夹爪位置记录进来。
grab_id1_deg = -18.0
grab_id2_deg = 37.7
grab_id3_deg = 20.0
grab_id4_deg = 40.0
grab_id5_deg = 0.0

# 夹球位置快速修正。单位为厘米，建议每次只改0.5；0表示使用上面的基础姿态。
# 正数向机器人前方伸远，负数向机器人方向收近。
# 例：夹爪伸过球约1cm，改成-1.0；夹爪还差1cm够不到球，改成+1.0。
grab_forward_offset_cm = -1.0

# 正数向机器人左侧移动，负数向机器人右侧移动。
# 例：夹爪在球右边约0.5cm，需要向左修正时改成+0.5。
grab_lateral_offset_cm = 0.0

# 正数升高，负数降低。
# 例：夹爪比球低约0.5cm时改成+0.5；比球高约0.5cm时改成-0.5。
grab_height_offset_cm = 0.0

# 直接加到最终grab_id4_deg，单位为度，建议每次改5。
# 它不会改变ID6夹爪力度；如果方向变差，就改为相反符号测试。
grab_pitch_offset_deg = 0.0

# 收球姿态：夹住并抬离地面后，机械臂平滑收至该姿态，随后才允许车轮启动。
# 这是此前实机手动记录并验证过的车轮启动前姿态。
carry_id1_deg = -11.3
carry_id2_deg = -18.3
carry_id3_deg = -45.0
carry_id4_deg = 51.8
carry_id5_deg = 0.1

# ID6夹爪开合量。保持现有夹球力度时不要修改。
# open为正表示在当前夹爪角度基础上打开；close为负表示闭合。
gripper_open_delta_deg = 60.0
gripper_close_delta_deg = -60.0

# 从抬球位置收至carry姿态所需时间，以及到位后的额外稳定时间，单位毫秒。
# duration越大收臂越慢、更柔和；过小会显得突然。settle只增加到位等待时间。
carry_duration_ms = 2000
carry_settle_ms = 500

# 视觉停车参数。检测框尺寸取网球检测框宽、高中的较大值。
# 目标尺寸：增大表示机器人需要更靠近球才停车；减小表示离球更远就停车。
ball_stop_size_px = 155

# 尺寸容差：球框尺寸落在“目标值±容差”内才满足距离条件。
# 当前155±5即允许150～160像素。增大更容易停车但前后位置误差更大；
# 减小停车距离更一致，但摄像头检测抖动时可能前后反复调整。
ball_stop_tolerance_px = 5

# 球心左右容差：球中心与画面目标中心的差不超过该像素值才满足对正条件。
# 当前允许左右偏差±30像素。增大更容易开始抓球但左右误差更大；
# 减小会对得更正，但低帧率或检测抖动时可能左右反复旋转。
ball_center_tolerance_px = 30

# 连续多少个检测帧同时满足“距离”和“左右对正”后，才启动机械臂。
# 增大可过滤偶发误检但等待更久。Starry真实闭环约2.3fps，2帧约需0.9秒。
ball_stable_frames = 2
