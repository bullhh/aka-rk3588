# LeKiwi C++ 抓取调参文件。
# x/y 单位是米，对应 Desktop-Wanderer 机械臂二维逆运动学平面。
# 如果夹爪伸过球、在球前方闭合，减小 grab_x。
# 如果夹爪够不到球，增大 grab_x。
# 如果夹爪太高或太低，调整 grab_y。

home_x = 0.0989
home_y = 0.1250

pre_grab_x = 0.0900
pre_grab_y = 0.1211
grab_x = 0.0900
grab_y = -0.0800

lift_x = -0.1000
lift_y = 0.2000

shoulder_pan_delta = -12
gripper_open_delta = 60
gripper_close_delta = -60
wrist_pick_pitch = 80
wrist_lift_pitch = -20

# 抓取时夹爪的腕部角度由 wrist_pick_pitch 控制。
# 如果夹爪不是尽量垂直向下，而是明显前倾或后仰，可以优先调整该值。
# 建议每次改 5，例如 80 -> 75 或 85，观察夹爪闭合时是否更贴合球。

# 是否在夹爪闭合后把肩部水平转动关节转回原位。
# 0 表示不转回，便于观察是否夹住；1 表示按 Desktop-Wanderer 原动作转回。
return_shoulder_pan = 1

# 视觉伺服停车尺寸。当前默认 155 像素，让车比 168~170 像素时稍远一点停车。
# 增大该值，车会离球更近再停；减小该值，车会离球更远就停。
ball_target_size = 155
ball_size_tolerance = 15

# 进入抓取前允许的球中心误差，单位是像素。
# StarryOS 真实 RKNN 闭环帧率约 2.3fps，架空测试时检测中心会在约 +/-30px
# 内抖动。这里允许短时间稳定停车后进入抓取，避免一直 BALL_FINE_LEFT/RIGHT。
ball_center_tolerance = 30
stable_frames = 2

# 完整闭环中，如果第一次没有夹住，程序会先回到视觉追球重新对准，
# 然后再使用下一组小偏移抓取，避免球被碰动后还在旧位置反复夹。
# 偏移顺序在代码中固定为：
# (0,0), (-0.005,0), (-0.010,0), (-0.015,0), (-0.020,0),
# (+0.010,0), (0,-0.010), (0,+0.010), (-0.015,-0.010), (-0.015,+0.010)。
# 如果某次偏移夹住了球，程序会把成功的 grab_x/grab_y/pre_grab_x/pre_grab_y 写回本文件。
